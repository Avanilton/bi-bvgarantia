-- ────────────────────────────────────────────────────
-- 1. INADIMPLÊNCIA

-- ────────────────────────────────────────────────────
SELECT b.idimovel, i.nomefantasia,
       DATE_FORMAT(b.dataVecto, '%Y-%m')   AS mesRef,
       SUM(IFNULL(b.total, 0))             AS valor,
       SUM(IFNULL(b.juros, 0))             AS juros,
       SUM(IFNULL(b.correcao, 0))          AS correcao,
       SUM(IFNULL(b.multa, 0))             AS multa,
       SUM(IFNULL(b.encargo, 0))           AS encargo,
       SUM(IFNULL(b.tarifaBancaria, 0))    AS tarifaBoleto
FROM TBBOLETO b
JOIN TBIMOVEL i ON i.idEmpresa = b.idEmpresa AND i.idimovel = b.idimovel
WHERE b.idEmpresa  = ?         -- ID_EMPRESA
  AND b.pago       = 0
  AND b.cancelado  = 0
  AND b.dataVecto  <= CURDATE()
GROUP BY b.idimovel, i.nomefantasia, mesRef
HAVING valor > 0;

-- 2. RECEBIMENTO (D-1)

SELECT b.idimovel, i.nomefantasia,
       DATE_FORMAT(b.dataPgto, '%Y-%m')    AS mesRef,
       SUM(IFNULL(b.total, 0))             AS valor,
       SUM(IFNULL(b.juros, 0))             AS juros,
       SUM(IFNULL(b.correcao, 0))          AS correcao,
       SUM(IFNULL(b.multa, 0))             AS multa,
       SUM(IFNULL(b.encargo, 0))           AS encargo,
       SUM(IFNULL(b.tarifaBancaria, 0))    AS tarifaBoleto
FROM TBBOLETO b
JOIN TBIMOVEL i ON i.idEmpresa = b.idEmpresa AND i.idimovel = b.idimovel
WHERE b.idEmpresa  = ?         -- ID_EMPRESA
  AND b.pago       = 1
  AND b.cancelado  = 0
  AND b.dataPgto   IS NOT NULL
  AND b.dataPgto   >= DATE_FORMAT(NOW(), '%Y-%m-01')
  AND b.dataPgto   <= DATE_SUB(CURDATE(), INTERVAL 1 DAY)
GROUP BY b.idimovel, i.nomefantasia, mesRef
HAVING valor > 0;

-- ────────────────────────────────────────────────────
-- 5. RECEITAS VARIÁVEIS

-- ────────────────────────────────────────────────────
SELECT b.idimovel, i.nomefantasia,
       DATE_FORMAT(b.dataPgto, '%Y-%m')    AS mesRef,
       SUM(IFNULL(b.correcao, 0))          AS correcao,
       SUM(IFNULL(b.encargo, 0))           AS encargo,
       SUM(IFNULL(b.juros, 0))             AS juros,
       SUM(IFNULL(b.multa, 0))             AS multa,
       SUM(IFNULL(b.tarifaBancaria, 0))    AS tarifaBoleto,
       (SUM(IFNULL(b.correcao, 0)) + SUM(IFNULL(b.encargo, 0))
        + SUM(IFNULL(b.juros, 0)) + SUM(IFNULL(b.multa, 0))
        + SUM(IFNULL(b.tarifaBancaria, 0))) AS valor
FROM TBBOLETO b
JOIN TBIMOVEL i ON i.idEmpresa = b.idEmpresa AND i.idimovel = b.idimovel
WHERE b.idEmpresa  = ?         -- ID_EMPRESA
  AND b.pago       = 1
  AND b.cancelado  = 0
  AND b.dataPgto   IS NOT NULL
  AND b.dataPgto   < CURDATE()
GROUP BY b.idimovel, i.nomefantasia, mesRef
HAVING valor > 0;


-- ────────────────────────────────────────────────────
-- 6. LISTA DE CONDOMÍNIOS

-- ────────────────────────────────────────────────────
SELECT idimovel, nomefantasia
FROM TBIMOVEL
WHERE idEmpresa = ?            -- ID_EMPRESA
ORDER BY nomefantasia;


========================================================
  OBSERVAÇÕES
========================================================

Tabelas utilizadas:
  - TBBOLETO  : boletos (principal)
  - TBIMOVEL  : condomínios/imóveis

Colunas relevantes do TBBOLETO:
  - idboleto, idEmpresa, idimovel, idcliente
  - pago        (0 = não pago, 1 = pago)
  - cancelado   (0 = ativo, 1 = cancelado)
  - origem      (5 = Amigável, 6 = Jurídico)
  - total       = valor principal do boleto
  - dataVecto   = data de vencimento
  - dataPgto    = data de pagamento
  - juros, correcao, multa, encargo, tarifaBancaria
